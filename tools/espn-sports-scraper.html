Run it from code

Use your Apify API token. The call waits for the run and returns the rows.

curl -X POST "https://api.apify.com/v2/acts/panix~espn-sports-scraper/run-sync-get-dataset-items?token=$APIFY_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"leagues":["nba","nfl","mlb","premier league"],"dateFrom":"yesterday","dateTo":"tomorrow","maxItems":30}'
from apify_client import ApifyClient

client = ApifyClient("APIFY_TOKEN")
run = client.actor("panix/espn-sports-scraper").call(run_input={
    "leagues": [
        "nba",
        "nfl",
        "mlb",
        "premier league"
    ],
    "dateFrom": "yesterday",
    "dateTo": "tomorrow",
    "maxItems": 30
})

for item in client.dataset(run["defaultDatasetId"]).iterate_items():
    print(item)
import { ApifyClient } from 'apify-client';

const client = new ApifyClient({ token: process.env.APIFY_TOKEN });
const run = await client.actor('panix/espn-sports-scraper').call({
  "leagues": [
    "nba",
    "nfl",
    "mlb",
    "premier league"
  ],
  "dateFrom": "yesterday",
  "dateTo": "tomorrow",
  "maxItems": 30
});

const { items } = await client.dataset(run.defaultDatasetId).listItems();
console.log(items);
