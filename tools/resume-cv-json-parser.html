Run it from code

Use your Apify API token. The call waits for the run and returns the rows.

curl -X POST "https://api.apify.com/v2/acts/panix~resume-cv-json-parser/run-sync-get-dataset-items?token=$APIFY_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"documentUrls":["https://api.apify.com/v2/key-value-stores/gZUDoFSN6sExqUTEm/records/sample-cv.pdf"]}'
from apify_client import ApifyClient

client = ApifyClient("APIFY_TOKEN")
run = client.actor("panix/resume-cv-json-parser").call(run_input={
    "documentUrls": [
        "https://api.apify.com/v2/key-value-stores/gZUDoFSN6sExqUTEm/records/sample-cv.pdf"
    ]
})

for item in client.dataset(run["defaultDatasetId"]).iterate_items():
    print(item)
import { ApifyClient } from 'apify-client';

const client = new ApifyClient({ token: process.env.APIFY_TOKEN });
const run = await client.actor('panix/resume-cv-json-parser').call({
  "documentUrls": [
    "https://api.apify.com/v2/key-value-stores/gZUDoFSN6sExqUTEm/records/sample-cv.pdf"
  ]
});

const { items } = await client.dataset(run.defaultDatasetId).listItems();
console.log(items);
