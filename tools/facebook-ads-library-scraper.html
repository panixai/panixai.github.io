Run it from code

Use your Apify API token. The call waits for the run and returns the rows.

curl -X POST "https://api.apify.com/v2/acts/panix~facebook-ads-library-scraper/run-sync-get-dataset-items?token=$APIFY_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"searchTerms":["running shoes"],"countries":["US"],"maxAdsPerSource":30,"proxyConfiguration":{"useApifyProxy":true,"apifyProxyGroups":["RESIDENTIAL"]}}'
from apify_client import ApifyClient

client = ApifyClient("APIFY_TOKEN")
run = client.actor("panix/facebook-ads-library-scraper").call(run_input={
    "searchTerms": [
        "running shoes"
    ],
    "countries": [
        "US"
    ],
    "maxAdsPerSource": 30,
    "proxyConfiguration": {
        "useApifyProxy": True,
        "apifyProxyGroups": [
            "RESIDENTIAL"
        ]
    }
})

for item in client.dataset(run["defaultDatasetId"]).iterate_items():
    print(item)
import { ApifyClient } from 'apify-client';

const client = new ApifyClient({ token: process.env.APIFY_TOKEN });
const run = await client.actor('panix/facebook-ads-library-scraper').call({
  "searchTerms": [
    "running shoes"
  ],
  "countries": [
    "US"
  ],
  "maxAdsPerSource": 30,
  "proxyConfiguration": {
    "useApifyProxy": true,
    "apifyProxyGroups": [
      "RESIDENTIAL"
    ]
  }
});

const { items } = await client.dataset(run.defaultDatasetId).listItems();
console.log(items);
