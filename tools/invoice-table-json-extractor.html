Run it from code

Use your Apify API token. The call waits for the run and returns the rows.

curl -X POST "https://api.apify.com/v2/acts/panix~invoice-table-json-extractor/run-sync-get-dataset-items?token=$APIFY_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"documentUrls":["https://slicedinvoices.com/pdf/wordpress-pdf-invoice-plugin-sample.pdf"]}'
from apify_client import ApifyClient

client = ApifyClient("APIFY_TOKEN")
run = client.actor("panix/invoice-table-json-extractor").call(run_input={
    "documentUrls": [
        "https://slicedinvoices.com/pdf/wordpress-pdf-invoice-plugin-sample.pdf"
    ]
})

for item in client.dataset(run["defaultDatasetId"]).iterate_items():
    print(item)
import { ApifyClient } from 'apify-client';

const client = new ApifyClient({ token: process.env.APIFY_TOKEN });
const run = await client.actor('panix/invoice-table-json-extractor').call({
  "documentUrls": [
    "https://slicedinvoices.com/pdf/wordpress-pdf-invoice-plugin-sample.pdf"
  ]
});

const { items } = await client.dataset(run.defaultDatasetId).listItems();
console.log(items);
