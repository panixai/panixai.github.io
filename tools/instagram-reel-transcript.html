Run it from code

Use your Apify API token. The call waits for the run and returns the rows.

curl -X POST "https://api.apify.com/v2/acts/panix~instagram-reel-transcript/run-sync-get-dataset-items?token=$APIFY_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"reels":["https://www.instagram.com/reel/DdUAMWNBktR/","https://www.instagram.com/reel/DehZj9wggt0/"],"proxyConfiguration":{"useApifyProxy":true}}'
from apify_client import ApifyClient

client = ApifyClient("APIFY_TOKEN")
run = client.actor("panix/instagram-reel-transcript").call(run_input={
    "reels": [
        "https://www.instagram.com/reel/DdUAMWNBktR/",
        "https://www.instagram.com/reel/DehZj9wggt0/"
    ],
    "proxyConfiguration": {
        "useApifyProxy": True
    }
})

for item in client.dataset(run["defaultDatasetId"]).iterate_items():
    print(item)
import { ApifyClient } from 'apify-client';

const client = new ApifyClient({ token: process.env.APIFY_TOKEN });
const run = await client.actor('panix/instagram-reel-transcript').call({
  "reels": [
    "https://www.instagram.com/reel/DdUAMWNBktR/",
    "https://www.instagram.com/reel/DehZj9wggt0/"
  ],
  "proxyConfiguration": {
    "useApifyProxy": true
  }
});

const { items } = await client.dataset(run.defaultDatasetId).listItems();
console.log(items);
